# Rotas do serviço de Equipes Técnicas

# --- Equipes ---

# [GET] /api/v1/equipes
# Lista todas as equipes.
# Permite filtrar por status e/ou especialidade com query parameters.
# Ex: /api/v1/equipes?status=ATIVA&especialidade=hidraulica

# [GET] /api/v1/equipes/<id>
# Busca uma equipe específica pelo seu ID.

#[GET] /api/v1/equipes?status=ATIVA
# Retorna todas as equipes que estão ativas.

#[GET] http://localhost:5000/api/v1/equipes/<int:id>/status
# Busca o status de uma equipe específica pelo seu ID.

#[GET] /api/v1/equipes/status/ativa/<int:id>
# Retorna equipe se a equipe com id está ativa.


# [POST] /api/v1/equipes
# Cria uma nova equipe.


# [PUT] /api/v1/equipes/<int:id>
# Atualiza todos os dados de uma equipe.

# [PATCH] /api/v1/equipes/<int:id>/<status>
# Atualiza parcialmente o status de uma equipe (ex: de ATIVA para INATIVA).


# [DELETE] /api/v1/equipes/<int:id>
# Deleta uma equipe (se não tiver ordens de serviço) ou a desativa.


# --- Membros ---
# [GET] /api/v1/membros
# Lista todos os membros de todas as equipes.

# [POST] /api/v1/membros
# Cria um novo membro e o associa a uma equipe.

# [GET] /api/v1/membros/<id>
# Busca um membro específico pelo seu ID.

# [PUT] /api/v1/membros/<id>
# Atualiza os dados de um membro.

# [DELETE] /api/v1/membros/<id>
# Remove um membro.
